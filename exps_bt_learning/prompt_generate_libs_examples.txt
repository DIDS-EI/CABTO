
Example 1:
initial_state = {"IsHandEmpty()"}
objects = {"coffeecup","apple","table","orange"}
goals = ["On(coffeecup,table)","On(coffeecup,table)&On(apple,table)","On(coffeecup,table)&On(apple,table)&On(orange,table)"]

```python
class PutAOnB(OGAction):
    can_be_expanded = True
    num_args = 2
    valid_args = list(itertools.product(["coffeecup","apple","orange"], ["table"]))
    def __init__(self, *args):
        super().__init__(*args)

    @classmethod
    def get_info(cls,*arg):
        info = {}
        info["pre"]={"IsHandEmpty()"}
        info["add"]={f"On({arg[0]},{arg[1]})"}
        # 删除所有 On(arg[0], 其他位置) 的状态
        all_locations = {args[1] for args in cls.valid_args}
        info["del_set"] = {f"On({arg[0]},{loc})" for loc in all_locations if loc != arg[1]}
        info["cost"] = 1
        return info

```

Example 2:
initial_state = {"IsHandEmpty(robot1)","IsHandEmpty(robot2)","On(robot1,area1)","On(robot2,area2)","On(x,area1)","On(y,area2)","Robot1CanOnlyDoInArea1()","Robot2CanOnlyDoInArea2()"}
objects = {"x","y","place","area1","area2"}
goals = ["On(x,place)","On(x,place)&On(y,place)"]

```python
class PickUp1(OGAction):
    can_be_expanded = True
    num_args = 1
    # robot1 can only pick up objects in area1: x
    valid_args = ["x"]

    def __init__(self, *args):
        super().__init__(*args)

    @classmethod
    def get_info(cls, *arg):
        info = {}
        # Preconditions: Robot's hand is empty, object is in area1, robot is in area1, robot1 restriction
        info["pre"] = {f"IsHandEmpty(robot1)", f"On({arg[0]},area1)", f"On(robot1,area1)", f"Robot1CanOnlyDoInArea1()"}
        # Added: Robot is holding the object
        info["add"] = {f"Holding(robot1,{arg[0]})"}
        # Deleted: Robot's hand is empty, object is in area1
        info["del_set"] = {f"IsHandEmpty(robot1)", f"On({arg[0]},area1)"}
        info["cost"] = 1
        return info
```
```python
class Drop1(OGAction):
    can_be_expanded = True
    num_args = 2
    # robot1 can drop 'x' into 'place' or 'area1' (since it's restricted to area1, it must be able to move it out of area1 if necessary)
    # The problem implies movement between areas/places is needed for the goal, so we assume 'place' is accessible from 'area1' for drop/put
    # Given the constraint Robot1CanOnlyDoInArea1(), the *drop* must be done *in* area1 or *to* a location accessible *from* area1. 
    # Let's assume 'place' is the destination, and 'area1' is a potential temporary drop location within its zone.
    valid_args = list(itertools.product(["x"], ["place", "area1"])) 

    def __init__(self, *args):
        super().__init__(*args)

    @classmethod
    def get_info(cls, *arg):
        info = {}
        # Preconditions: Robot is holding the object, robot is in area1, robot1 restriction
        info["pre"] = {f"Holding(robot1,{arg[0]})", f"On(robot1,area1)", f"Robot1CanOnlyDoInArea1()"}
        # Added: Robot's hand is empty, object is on the destination location
        info["add"] = {f"IsHandEmpty(robot1)", f"On({arg[0]},{arg[1]})"}
        # Deleted: Robot is holding the object
        info["del_set"] = {f"Holding(robot1,{arg[0]})"}
        info["cost"] = 1
        return info
```
