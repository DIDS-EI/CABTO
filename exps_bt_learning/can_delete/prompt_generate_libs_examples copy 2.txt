class PutAOnB(OGAction):
    can_be_expanded = True
    num_args = 2
    valid_args = list(itertools.product(["coffeecup","apple","orange"], ["table"]))
    def __init__(self, *args):
        super().__init__(*args)

    @classmethod
    def get_info(cls,*arg):
        info = {}
        info["pre"]={"IsHandEmpty()"}
        info["add"]={f"On({arg[0]},{arg[1]})"}
        # 删除所有 On(arg[0], 其他位置) 的状态
        all_locations = {args[1] for args in cls.valid_args}
        info["del_set"] = {f"On({arg[0]},{loc})" for loc in all_locations if loc != arg[1]}
        info["cost"] = 1
        return info

start_state = {"IsHandEmpty(robot1)","IsHandEmpty(robot2)","On(robot1,area1)","On(robot2,area2)","On(x,area1)","On(y,area2)","Robot1CanOnlyDoInArea1()","Robot2CanOnlyDoInArea2()"}

import itertools
from typing import Set, Dict, Any, List, Tuple
ROBOTS = {"robot1", "robot2"}
CARRY_ITEMS = {"x", "y"}
LOCATIONS = {"area1", "area2", "place"}
ALL_OBJECTS = CARRY_ITEMS.union(LOCATIONS)
ALL_ROBOTS = ROBOTS
ALL_ENTITIES = ALL_ROBOTS.union(ALL_OBJECTS)
ALL_MOVEABLE_ITEMS = CARRY_ITEMS.union(ROBOTS)

class Move(OGAction):
    can_be_expanded = True
    num_args = 3 # Move(robot, from_loc, to_loc)
    # robot can move between any two locations
    valid_args = list(itertools.product(ROBOTS, LOCATIONS, LOCATIONS))

    @classmethod
    def get_info(cls, *arg) -> Dict[str, Any]:
        robot, from_loc, to_loc = arg
        info = {}
        info["pre"] = {f"On({robot},{from_loc})"}
        info["add"] = {f"On({robot},{to_loc})"}
        info["del_set"] = {f"On({robot},{from_loc})"}
        info["cost"] = 1
        return info

class Pickup(OGAction):
    can_be_expanded = True
    num_args = 2 # Pickup(robot, item)
    # Robots can pick up items when On(robot, location) and On(item, location)
    # The location is implicit and inferred from the state
    valid_args = list(itertools.product(ROBOTS, CARRY_ITEMS))

    @classmethod
    def get_info(cls, *arg) -> Dict[str, Any]:
        robot, item = arg
        # Precondition check needs location, but here we define the state change.
        # The location-specific precondition must be checked by the planner.
        # We assume the robot is at the item's location.

        # Determine possible locations for the item and robot
        # In this simple model, we must consider all locations.
        all_item_locations = LOCATIONS

        pre_conditions = {f"IsHandEmpty({robot})"}

        # We need to express that the robot is at the same location as the item.
        # Since we don't know the location 'L' here, we must make the planner check 
        # for an 'L' such that On(robot, L) AND On(item, L).
        # We introduce a placeholder for the planner's benefit:
        pre_conditions.add("AtSameLocation(robot, item)") # This is a conceptual check for the planner

        # For the provided start_state, we can infer some constraints.
        # robot1 <-> area1, robot2 <-> area2. x, y are at area1/area2 initially.
        if robot == "robot1":
            pre_conditions.add(f"RobotCanOnlyDoIn(robot1,area1)")
        elif robot == "robot2":
            pre_conditions.add(f"RobotCanOnlyDoIn(robot2,area2)")

        # The core state changes:
        info = {}
        info["pre"] = pre_conditions # Requires IsHandEmpty(robot) AND On(robot, L) AND On(item, L)
        info["add"] = {f"Holding({robot},{item})"}

        # Deleting all On(item, L) facts
        info["del_set"] = {f"On({item},{loc})" for loc in all_item_locations}
        info["del_set"].add(f"IsHandEmpty({robot})")

        info["cost"] = 1
        return info
```

class Putdown(OGAction):
    can_be_expanded = True
    num_args = 3 # Putdown(robot, item, location)
    # The robot must be holding the item and be at the location
    valid_args = list(itertools.product(ROBOTS, CARRY_ITEMS, LOCATIONS))

    @classmethod
    def get_info(cls, *arg) -> Dict[str, Any]:
        robot, item, location = arg

        pre_conditions = {f"Holding({robot},{item})", f"On({robot},{location})"}

        # Robot location constraints for action execution
        if robot == "robot1":
            pre_conditions.add(f"RobotCanOnlyDoIn(robot1,area1)")
        elif robot == "robot2":
            pre_conditions.add(f"RobotCanOnlyDoIn(robot2,area2)")

        info = {}
        info["pre"] = pre_conditions
        info["add"] = {f"On({item},{location})", f"IsHandEmpty({robot})"}
        info["del_set"] = {f"Holding({robot},{item})"}
        info["cost"] = 1
        return info
```
