        super().__init__(*args)

    @classmethod
    def get_info(cls, *arg):
        obj, place = arg
        info = {}
        # Preconditions: holding the object, and if place is cabinet/fridge must be opened
        preconds = {f"Holding({obj})"}
        if place in ["cabinet", "fridge"]:
            preconds.add(f"IsOpened({place})")
        info["pre"] = preconds
        info["add"] = {"IsHandEmpty()"}
        # Add On or In depending on place: if place is cabinet/fridge, In(obj,place), else On(obj,place)
        if place in ["cabinet", "fridge"]:
            info["add"].add(f"In({obj},{place})")
            # Remove possible prior On predicates of obj on table or breakfast_table
            info["del_set"] = {f"Holding({obj})"}
            info["del_set"].update({f"On({obj},table)", f"On({obj},breakfast_table)"})
            info["del_set"].discard(f"In({obj},{place})")  # Don't delete what we add
        else:
            info["add"].add(f"On({obj},{place})")
            # Remove possible prior In predicates obj in fridge or cabinet
            info["del_set"] = {f"Holding({obj})"}
            info["del_set"].update({f"In({obj},fridge)", f"In({obj},cabinet)"})
            info["del_set"].discard(f"On({obj},{place})")
        info["cost"] = 1
        return info
